class StrategyManager:
    def __init__(self):
        self.strategies = {
            'bullish': 'bullish_strategy.json',
            'bearish': 'bearish_strategy.json',
            'range': 'range_strategy.json',
            'default': 'default_conservative_strategy.json'
        }
    
    def get_current_strategy(self):
        """获取当前应使用的策略"""
        try:
            # 1. 尝试读取宏观趋势
            with open('data/current_trend.json', 'r') as f:
                trend_data = json.load(f)
            
            # 2. 检查时效性（3小时内有效）
            analysis_time = datetime.fromisoformat(trend_data['analysis_time'].replace('Z', '+00:00'))
            time_diff = datetime.now(timezone.utc) - analysis_time
            
            # 3. 检查置信度
            confidence = trend_data.get('confidence_percentage', 0)
            
            # 4. 决定使用哪个策略
            if time_diff.total_seconds() > 3 * 3600:  # 超过3小时
                logger.warning("宏观趋势数据过期，使用默认策略")
                return self.strategies['default']
            
            if confidence < 60:  # 置信度过低
                logger.warning(f"宏观趋势置信度过低({confidence}%)，使用默认策略")
                return self.strategies['default']
            
            # 5. 根据趋势选择策略
            trend = trend_data['market_trend']
            if trend in self.strategies:
                logger.info(f"使用{trend}策略，置信度{confidence}%")
                return self.strategies[trend]
            else:
                logger.warning(f"未知趋势类型{trend}，使用默认策略")
                return self.strategies['default']
                
        except Exception as e:
            # 任何异常都回退到默认策略
            logger.error(f"获取策略失败: {e}，使用默认策略")
